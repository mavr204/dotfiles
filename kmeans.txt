import numpy as np

# Sample data
X = np.array([[1, 2], [1, 4], [1, 0],
              [10, 2], [10, 4], [10, 0]])

k = 2
centroids = X[:k]  # Initialize centroids (first k points)

for _ in range(5):  # Fixed number of iterations
    labels = []
    for x in X:
        distances = [np.linalg.norm(x - c) for c in centroids]
        labels.append(np.argmin(distances))
    
    centroids = np.array([
        X[np.array(labels) == i].mean(axis=0)
        for i in range(k)
    ])

print("Final Centroids:\n", centroids)
print("Labels:", labels)
